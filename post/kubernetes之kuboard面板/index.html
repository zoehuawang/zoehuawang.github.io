kuboard在k8s中一件部署 

#  初始化sql：
#  CREATE DATABASE kuboard DEFAULT CHARACTER SET = 'utf8mb4' DEFAULT COLLATE = 'utf8mb4_unicode_ci';
#  create user 'kuboard'@'%' identified by 'kuboardpwd';
#  grant all privileges on kuboard.* to 'kuboard'@'%';
#  FLUSH PRIVILEGES;
#
#  默认用户密码：admin / Kuboard123
---
apiVersion: apps/v1
kind: Deployment
metadata:
  labels:
    app: kuboard
  name: kuboard
spec:
  replicas: 1
  selector:
    matchLabels:
      app: kuboard
  template:
    metadata:
      labels:
        app: kuboard
    spec:
      containers:
        - env:
            - name: DB_DRIVER
              value: "org.mariadb.jdbc.Driver"
            - name: DB_URL
              value: "jdbc:mariadb://mysql-0.mysql.middleware:3306/kuboard?serverTimezone=Asia/Shanghai"
            - name: DB_USERNAME
              value: "kubboard"
            - name: DB_PASSWORD
              value: "kuboardpwd"
          image: swr.cn-east-2.myhuaweicloud.com/kuboard/kuboard:v4
          name: kuboard
          ports:
            - containerPort: 80
              protocol: TCP
      restartPolicy: Always
---
apiVersion: v1
kind: Service
metadata:
  labels:
    app: kuboard
  name: kuboard
spec:
  ports:
    - name: '80'
      nodePort: 30000
      port: 80
      protocol: TCP
      targetPort: 80
  selector:
    app: kuboard
  type: NodePort
